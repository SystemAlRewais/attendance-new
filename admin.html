async function updateAdminSettingDirect(key, value) { const statusMsg = document.getElementById('adminStatusMsg'); statusMsg.style.color = "#38bdf8"; statusMsg.textContent = "⏳ جاري تحديث الحالة في السحابة..."; try { // جلب الإعدادات الحالية من الشيت الأول لضمان عدم ضياع القيمة الأخرى let currentSettings = { isAttendanceOpen: "true", isRegistrationOpen: "true", currentSubject: document.getElementById('adminCurrentSubject').value || "المحاضرة الأولى" }; try { const res = await fetch(`${sheetDBUrl}?sheet=Setting`, { cache: "no-store" }); const data = await res.json(); if (Array.isArray(data) && data.length > 0) { currentSettings = data[0]; } } catch(e) {} // تحديث المفتاح المطلوب فقط (سواء الحضور أو التسجيل أو العنوان) currentSettings[key] = value; if(key !== 'currentSubject') { currentSettings.currentSubject = document.getElementById('adminCurrentSubject').value || currentSettings.currentSubject; } // مسح وإعادة كتابة الإعدادات بالقيم المنفصلة السليمة await fetch(`${sheetDBUrl}?sheet=Setting`, { method: 'DELETE', body: JSON.stringify({ query: { id: "*" } }) }).catch(()=>{}); await fetch(`${sheetDBUrl}?sheet=Setting`, { method: 'POST', headers: { 'Accept': 'application/json', 'Content-Type': 'application/json' }, body: JSON.stringify({ data: [currentSettings] }) }); updateUIButtonsState(currentSettings.isAttendanceOpen, currentSettings.isRegistrationOpen); statusMsg.style.color = "var(--accent-green)"; statusMsg.textContent = `✅ تم تحديث الحالة المطلوبة بنجاح!`; } catch (e) { console.error(e); statusMsg.style.color = "var(--accent-red)"; statusMsg.textContent = "❌ حدث خطأ أثناء التحديث."; } }
